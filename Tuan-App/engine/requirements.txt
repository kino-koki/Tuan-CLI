# Tuan-App 打包引擎的第三方依赖清单
# ------------------------------------
# 与仓库根 pyproject.toml 的 [project.dependencies] 核心集保持同步。
# 引擎代码本身在 `tuan_cli/`（由 scripts/sync_engine_bundle.py 同步），
# 本文件只负责装**第三方依赖** —— 应用运行时以 cwd=engine/ 直接导入 tuan_cli，
# 依赖则从这个虚拟环境里取。
#
# 为什么要单独一份：打包后的 Windows 机器通常没有 Python、或者只有
# Microsoft Store 的引导存根；即使用户装了真 Python，也不会自动带这些包。
# 缺包时引擎会直接在 `import rich` 处崩掉（ModuleNotFoundError），
# 用户看到的是一个看不懂的 Python 栈。
#
# 安装：python scripts/install_engine_env.py
# 校验：python scripts/install_engine_env.py --check

# 终端 UI 输出（tuan_cli/ui/plain_console.py 模块级导入）
rich>=13.0
# 模型传输（tuan_cli/models/anthropic.py 模块级导入）
httpx>=0.27
# 默认 TUI（缺失时进交互界面会 NameError）
prompt_toolkit>=3.0
# Python < 3.11 无内置 tomllib
tomli>=2.0; python_version < "3.11"
# config/loader.py 的 YAML 解析（不装会静默退回极简解析器，语义被改写）
pyyaml>=6.0
# OpenAI 兼容传输层（大多数用户的主路径）
openai>=1.40
# MCP 子系统配置模型
pydantic>=2.0
typing_extensions>=4.0
# crypto 引擎的 AES-GCM 实现
cryptography>=41.0
